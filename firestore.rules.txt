rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    
    // Helper Function: ตรวจสอบว่าเป็น Admin หรือไม่ (ตรวจสอบจาก Custom Claims หรือ Email สำรอง)
    function isAdmin() {
      return request.auth != null && (
        request.auth.token.admin == true || 
        request.auth.token.email == "bentsbac@gmail.com" || 
        request.auth.token.email == "kwan.oneself@gmail.com"
      );
    }

    // Helper Function: ตรวจสอบว่าตลาดซื้อขาย/จัดทีม ยังเปิดอยู่หรือไม่
    function isGameweekOpen() {
      let sysConfig = get(/databases/$(database)/documents/public_data/system_config).data;
      return sysConfig.isMarketOpen == true;
    }

    // 👤 ระบบผู้ใช้งาน (Users & Sub-collections)
    match /users/{userId} {
      // ให้อ่านได้ทุกคนที่ล็อกอิน (เผื่อดูสถิติเพื่อน) แต่แก้ไขได้เฉพาะเจ้าของ
      // ป้องกันการแก้ไข Balls ให้ติดลบ
      allow read: if request.auth != null;
      allow write: if request.auth != null && (
        (request.auth.uid == userId && (!request.resource.data.keys().hasAny(['balls']) || request.resource.data.balls >= 0))
        || isAdmin()
      );
      
      // การเข้าถึง Sub-collection ด้านในของ users
      match /{document=**} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && (request.auth.uid == userId || isAdmin());
      }
      
      // ให้เฉพาะ Admin หรือตัวเองเท่านั้นที่เขียน gameweek_history ได้
      match /gameweek_history/{gameweekId} {
        allow read: if request.auth != null;
        allow write: if request.auth != null && (request.auth.uid == userId || isAdmin());
      }
    }

    // ⚽ ระบบจัดการนักเตะ (Database)
    match /players/{document=**} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // 🛡️ ระบบจัดการทีม (Teams - Clubs)
    match /teams/{document=**} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // 🏆 ระบบจัดการการแข่งขัน
    match /matches/{document=**} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // 🎯 ระบบเควสต์และสปอนเซอร์
    match /quests/{document=**} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }

    // 🎁 ระบบร้านค้า ของรางวัล
    match /rewards/{document=**} {
      allow read: if request.auth != null;
      allow write: if isAdmin();
    }
    
    // 💳 ระบบประวัติการทำรายการ Statement
    match /transactions/{document=**} {
      allow read, write: if false; // Should not be accessible from root. Transactions are inside users/{userId}
    }

    // ⚙️ ระบบตั้งค่า
    match /settings/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // 🧩 ระบบไฟล์เสริม (Artifacts) ที่ระบบ Admin จำเป็นต้องใช้งาน
    match /artifacts/{appId}/public/data/managers/{managerId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    match /artifacts/{appId}/users/{userId} {
      // 🚨 ล็อกการเซฟทีม (Squad) เมื่อหมดเวลา
      match /game_data/squad {
        allow read: if true;
        allow write: if request.auth != null && (isAdmin() || (request.auth.uid == userId && isGameweekOpen()));
      }
      
      // การเข้าถึง Artifacts ไฟล์อื่นๆ ปล่อยตามปกติ แต่ห้ามซ้ำกับ squad
      match /{document=**} {
        allow read: if true;
        allow write: if request.auth != null && (request.auth.uid == userId || isAdmin()) 
          && !(request.path[7] == 'game_data' && request.path[8] == 'squad');
      }
    }
    match /artifacts/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // 📊 ระบบสถิติส่วนกลาง และ การตั้งค่าโฆษณา
    match /public_data/system_config {
      allow read: if true;
      allow write: if isAdmin() || (
        request.auth != null && 
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['totalJoinedTeams'])
      );
    }
    match /public_data/ads_config {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    // 🏆 ความสำเร็จและฉายา
    match /public_data/achievements/list/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // 📋 กติกาเกม คะแนน และเงื่อนไข
    match /public_data/game_rules {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /public_data/scoring_rules {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /public_data/game_conditions {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    // 📅 ระบบสถานะแต่ละ Gameweek
    match /public_data/gameweeks/weeks/{gameweekId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // 📡 ระบบ Live Match (ควบคุมโดยแอดมิน)
    match /public_data/live_match {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    match /public_data/live_match/events/{eventId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // 📚 คลังข้อมูลอดีต (Historical Data)
    match /historical_players/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /historical_teams/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }
    match /historical_fixtures/{document=**} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    // 📊 ข้อมูลสถิติสดระหว่างสัปดาห์
    match /public_data/live_gameweek_stats/{playerId} {
      allow read: if true;
      allow write: if isAdmin();
    }
    
    // 🛡️ ประวัติการดึงข้อมูล (Admin Data)
    match /api_fetch_history/{document=**} {
      allow read: if isAdmin();
      allow write: if isAdmin();
    }

    // 💬 ระบบแชทส่วนรวม
    match /global_chat/{document=**} {
      allow read: if true;
      allow write: if request.auth != null && (request.resource.data.userId == request.auth.uid || isAdmin());
    }

    // 🤝 ระบบลีกส่วนตัว
    match /leagues/{document=**} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // 🎁 ระบบ Referral
    match /referrals/{document=**} {
      allow read: if request.auth != null;
      allow write: if request.auth != null;
    }

    // 🔒 ปิดกั้นการเข้าถึง Collection อื่นๆ นอกเหนือจากนี้ เพื่อความปลอดภัย
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
